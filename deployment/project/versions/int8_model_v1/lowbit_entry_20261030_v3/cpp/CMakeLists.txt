cmake_minimum_required(VERSION 3.16)
project(d_integer_entry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Icraft-XRT REQUIRED)
find_package(Icraft-HostBackend REQUIRED)
add_executable(d_integer_host integer_host.cpp)
target_link_libraries(d_integer_host PRIVATE Icraft::XRT Icraft::HostBackend)
target_compile_options(d_integer_host PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus)
add_executable(d_fold_scale fold_scale.cpp)
target_link_libraries(d_fold_scale PRIVATE Icraft::XRT Icraft::HostBackend)
target_compile_options(d_fold_scale PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus)
add_executable(d_audit_weights audit_weights.cpp)
target_link_libraries(d_audit_weights PRIVATE Icraft::XRT Icraft::HostBackend)
target_compile_options(d_audit_weights PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus)
