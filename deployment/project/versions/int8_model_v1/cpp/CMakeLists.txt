cmake_minimum_required(VERSION 3.16)
project(d_host_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Icraft-XRT REQUIRED)
find_package(Icraft-HostBackend REQUIRED)
add_executable(d_host_tools host_tools.cpp)
target_link_libraries(d_host_tools PRIVATE Icraft::XRT Icraft::HostBackend)
if(MSVC)
  target_compile_options(d_host_tools PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus)
endif()
